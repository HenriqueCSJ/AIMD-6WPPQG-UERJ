% S02 -- Das interacoes parametrizadas a energia, as forcas e ao movimento.
% Decomposicao ilustrativa de um campo de forca molecular convencional.
% U_FF e uma notacao generica; nao especifica o modelo Universal Force Field.
\begin{frame}[t,label=md-classica]{Dinâmica molecular clássica}
\lead{Um campo de força associa uma energia a cada geometria.}
\begin{conceptmath}
\[
 U_{\mathrm{FF}}=
 \underbrace{U_{\mathrm{lig}}+U_{\mathrm{ang}}+U_{\mathrm{tor}}}_{\text{interações ligadas}}
 +\underbrace{U_{\mathrm{vdW}}+U_{\mathrm{eletrost}}}_{\text{interações não ligadas}}
\]
\end{conceptmath}
{\fontsize{11}{14}\selectfont\color{Muted}
Ligações: alongar; ângulos: dobrar; torções: girar.\par
Não ligadas: van der Waals e eletrostática.\par}
\begin{conceptmath}
\[
 \F_I=-\nabla_I U_{\mathrm{FF}}(\R)
 \qquad
 M_I\,\mathbf a_I=\F_I
\]
\end{conceptmath}
{\fontsize{11}{14}\selectfont\color{Muted}
$\R$: posições; $\F_I$: força; $M_I$: massa; $\mathbf a_I$: aceleração do átomo $I$.\par}
\takeaway{Atualizar posições e velocidades; repetir para obter a trajetória.}
\source{https://www.faccts.de/docs/orca/6.1/manual/contents/modelchemistries/mm.html}{Referência: manual ORCA 6.1, seção 3.23 (Molecular Mechanics).}
\end{frame}

% S03 -- BOMD: o ciclo pratico do calculo eletronico e da propagacao nuclear.
% E_el exclui V_NN; E_BO inclui ambas as parcelas, sem dupla contagem.
% Energia e gradiente aproximados pelo metodo escolhido. Nucleos classicos.
\begin{frame}[t,label=aimd]{Dinâmica molecular \textit{ab initio}}
\lead{No ORCA: dinâmica de Born--Oppenheimer (BOMD).}
\begin{conceptmath}
\begin{gather*}
 E_{\mathrm{BO}}(\R)=
 \underbrace{E_{\mathrm{el}}(\R)}_{\text{energia eletrônica}}
 +\underbrace{V_{\mathrm{NN}}(\R)}_{\text{repulsão entre núcleos}}
 \\[2mm]
 \F_I=-\nabla_I E_{\mathrm{BO}}(\R)
 \qquad
 M_I\,\mathbf a_I=\F_I
\end{gather*}
\end{conceptmath}
\vspace{-7mm}
\begin{enumerate}
 \setlength{\itemsep}{1mm}
 \item Calcular energia e forças (ex.: DFT), com núcleos fixos.
 \item Atualizar posições e velocidades pelas forças obtidas.
 \item Repetir o cálculo para a nova geometria.
\end{enumerate}
\takeaway{Os núcleos seguem Newton; muda o cálculo das forças.}
\source{https://www.faccts.de/docs/orca/6.1/manual/contents/moleculardynamics/moldyn.html}{Referência: manual ORCA 6.1, seção 7.1 (Ab initio Molecular Dynamics).}
\end{frame}
